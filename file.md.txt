# rsa.py — Enhanced RSA with Extended Euclidean, prime validation, chunk encryption

def extended_gcd(a: int, b: int):
    if a == 0:
        return b, 0, 1
    gcd, x1, y1 = extended_gcd(b % a, a)
    return gcd, y1 - (b // a) * x1, x1


def mod_inverse(e: int, phi: int) -> int:
    gcd, x, _ = extended_gcd(e % phi, phi)
    if gcd != 1:
        raise ValueError(f"No modular inverse exists for e={e} and phi={phi}. Check your prime numbers.")
    return (x % phi + phi) % phi


def gcd(a: int, b: int) -> int:
    while b:
        a, b = b, a % b
    return a


def is_prime(n: int) -> bool:
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(n**0.5) + 1, 2):
        if n % i == 0:
            return False
    return True


def generate_keys(prime_a: int = 61, prime_b: int = 53):
    if not is_prime(prime_a):
        raise ValueError(f"{prime_a} is not a prime number.")
    if not is_prime(prime_b):
        raise ValueError(f"{prime_b} is not a prime number.")
    if prime_a == prime_b:
        raise ValueError("The two primes must be different.")

    modulus = prime_a * prime_b
    phi = (prime_a - 1) * (prime_b - 1)
    public_exponent = 17

    while gcd(public_exponent, phi) != 1:
        public_exponent += 2

    private_exponent = mod_inverse(public_exponent, phi)
    return (public_exponent, modulus), (private_exponent, modulus)


def _get_chunk_size(modulus: int) -> int:
    """Max bytes we can safely encrypt per chunk (must fit under modulus)."""
    return max(1, (modulus.bit_length() // 8) - 1)


def encrypt(text: str, public_key) -> list[int]:
    exponent, modulus = public_key
    chunk_size = _get_chunk_size(modulus)
    encoded = text.encode("utf-8")

    # Pad last chunk so every chunk is exactly chunk_size bytes.
    # We prepend a 0x01 marker byte so we can detect and strip padding on decrypt.
    remainder = len(encoded) % chunk_size
    padding_len = (chunk_size - remainder) % chunk_size
    # Store padding_len in a separate leading chunk so decrypt knows how much to strip
    padded = encoded + b"\x00" * padding_len

    encrypted_chunks: list[int] = []
    # First, store padding_len as its own encrypted value
    encrypted_chunks.append(pow(padding_len, exponent, modulus))

    for i in range(0, len(padded), chunk_size):
        chunk = padded[i : i + chunk_size]
        chunk_int = int.from_bytes(chunk, "big")
        if chunk_int >= modulus:
            raise ValueError(
                f"Chunk value {chunk_int} exceeds modulus {modulus}. "
                "Use larger prime numbers for longer messages."
            )
        encrypted_chunks.append(pow(chunk_int, exponent, modulus))

    return encrypted_chunks


def decrypt(cipher_values: list[int], private_key) -> str:
    exponent, modulus = private_key
    chunk_size = _get_chunk_size(modulus)

    if not cipher_values:
        raise ValueError("Empty ciphertext.")

    # First value is the padding length
    padding_len = pow(cipher_values[0], exponent, modulus)
    decrypted_bytes = bytearray()

    for value in cipher_values[1:]:
        dec_int = pow(value, exponent, modulus)
        chunk = dec_int.to_bytes(chunk_size, "big")
        decrypted_bytes.extend(chunk)

    # Strip the padding
    if padding_len > 0:
        decrypted_bytes = decrypted_bytes[:-padding_len]

    return decrypted_bytes.decode("utf-8")


def get_public_key(p: int = 61, q: int = 53):
    pub, _ = generate_keys(p, q)
    return pub


def get_private_key(p: int = 61, q: int = 53):
    _, priv = generate_keys(p, q)
    return priv


def rsa_algorithm(text: str, p: int = 61, q: int = 53) -> str:
    """Encrypt text and return a dash-separated string of integers."""
    try:
        if not text:
            raise ValueError("Message is empty.")
        pub_k, _ = generate_keys(p, q)
        encrypted_chunks = encrypt(text, pub_k)
        return "-".join(map(str, encrypted_chunks))
    except (TypeError, ValueError) as exc:
        return f"[ERROR: {exc}]"


def rsa_decrypt(text: str, p: int = 61, q: int = 53) -> str:
    """Decrypt a dash-separated string of integers back to original text."""
    try:
        if not text:
            raise ValueError("Ciphertext is empty.")
        if text.startswith("[ERROR"):
            raise ValueError(f"Cannot decrypt an error value: {text}")
        cipher_values = [int(v) for v in text.split("-") if v]
        if not cipher_values:
            raise ValueError("No valid integers found in ciphertext.")
        _, priv_k = generate_keys(p, q)
        return decrypt(cipher_values, priv_k)
    except (TypeError, ValueError) as exc:
        return f"[ERROR: {exc}]"
# hyprid.py — Enhanced Hybrid pipeline: Playfair → Vigenere → RSA
# Fixes: padding removal, error propagation, input validation

from playfair import DEFAULT_PLAYFAIR_KEY, playfair_algorithm, playfair_decrypt
from rsa import rsa_algorithm, rsa_decrypt
from vignere import DEFAULT_VIGENERE_KEY, vig_decrypt, vignere_algorithm

DEFAULT_HYBRID_PIPELINE = ("playfair", "vigenere", "rsa")


def _is_error(value: str) -> bool:
    return value.startswith("[ERROR")


def hybrid_encrypt(
    message: str,
    vig_key: str = DEFAULT_VIGENERE_KEY,
    rsa_pub_key=None,
    play_k: str = DEFAULT_PLAYFAIR_KEY,
) -> str:
    """
    Encrypt using pipeline: Playfair → Vigenere → RSA.
    Returns the encrypted string or '[ERROR: ...]' with a descriptive message.
    """
    del rsa_pub_key  # unused — RSA uses internally generated keypair

    if not message or not message.strip():
        return "[ERROR: Message is empty.]"

    # Step 1 — Playfair
    # playfair_algorithm returns "CIPHERTEXT|original_len"
    playfair_output = playfair_algorithm(message, play_k)
    if _is_error(playfair_output):
        return f"[ERROR: Playfair encrypt failed — {playfair_output}]"

    # Step 2 — Vigenere
    # Vigenere works on alpha chars; the "|len" suffix from Playfair must be
    # carried through intact. We encrypt only the cipher part and reattach suffix.
    if "|" in playfair_output:
        cipher_part, length_suffix = playfair_output.rsplit("|", 1)
    else:
        cipher_part, length_suffix = playfair_output, None

    vigenere_output = vignere_algorithm(cipher_part, vig_key)
    if _is_error(vigenere_output):
        return f"[ERROR: Vigenere encrypt failed — {vigenere_output}]"

    # Reattach length suffix so decrypt can unpad precisely
    combined = f"{vigenere_output}|{length_suffix}" if length_suffix is not None else vigenere_output

    # Step 3 — RSA
    rsa_output = rsa_algorithm(combined)
    if _is_error(rsa_output):
        return f"[ERROR: RSA encrypt failed — {rsa_output}]"

    return rsa_output


def hybrid_decrypt(
    cipher: str,
    vig_key: str = DEFAULT_VIGENERE_KEY,
    rsa_priv_key=None,
    play_k: str = DEFAULT_PLAYFAIR_KEY,
) -> str:
    """
    Decrypt using reverse pipeline: RSA → Vigenere → Playfair.
    Returns the original plaintext or '[ERROR: ...]'.
    """
    del rsa_priv_key  # unused

    if not cipher or not cipher.strip():
        return "[ERROR: Ciphertext is empty.]"

    # Step 1 — RSA decrypt
    rsa_output = rsa_decrypt(cipher)
    if _is_error(rsa_output):
        return f"[ERROR: RSA decrypt failed — {rsa_output}]"

    # Step 2 — Vigenere decrypt
    # Split off the "|length" suffix before passing to Vigenere
    if "|" in rsa_output:
        vigenere_input, length_suffix = rsa_output.rsplit("|", 1)
    else:
        vigenere_input, length_suffix = rsa_output, None

    vigenere_output = vig_decrypt(vigenere_input, vig_key)
    if _is_error(vigenere_output):
        return f"[ERROR: Vigenere decrypt failed — {vigenere_output}]"

    # Reattach length suffix for Playfair to use during unpadding
    playfair_input = f"{vigenere_output}|{length_suffix}" if length_suffix is not None else vigenere_output

    # Step 3 — Playfair decrypt (removes padding precisely using embedded length)
    playfair_output = playfair_decrypt(playfair_input, play_k)
    if _is_error(playfair_output):
        return f"[ERROR: Playfair decrypt failed — {playfair_output}]"

    return playfair_output
# enhancement.py — Compatibility wrapper around the enhanced hybrid pipeline
# Kept for backward compatibility with legacy_crypto_loader.py

from hyprid import hybrid_decrypt, hybrid_encrypt


def enhanced_algorithm(data: str) -> str:
    """
    Hybrid encrypt: Playfair → Vigenere → RSA.
    Returns encrypted string or '[ERROR: ...]'.
    """
    if not isinstance(data, str):
        return "[ERROR: Input must be a string.]"
    return hybrid_encrypt(data)


def enhance_decrypt(data: str) -> str:
    """
    Hybrid decrypt: RSA → Vigenere → Playfair.
    Returns original plaintext or '[ERROR: ...]'.
    """
    if not isinstance(data, str):
        return "[ERROR: Input must be a string.]"
    return hybrid_decrypt(data)
